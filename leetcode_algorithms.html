first_true — 二分法统一框架

把区间 [lo, hi) 分成前半段 False、后半段 True，找 第一个 True 的位置。全 False 返回 hi，全 True 返回 lo。
def first_true(lo, hi, predicate):
    while lo < hi:
        mid = lo + hi >> 1        # 向下取整，避免死循环
        if predicate(mid):
            hi = mid              # mid 满足，答案在 [lo, mid]
        else:
            lo = mid + 1          # mid 不满足，答案在 [mid+1, hi)
    return lo                     # lo == hi，即答案

def searchRotated(nums, target):
    n = len(nums)
    rot = first_true(0, n, lambda i: nums[i] < nums[0])   # Step1
    v   = first_true(0, n, lambda v: nums[(v + rot) % n] >= target)  # Step2
    real = (v + rot) % n
    return real if v < n and nums[real] == target else -1

删除子数组的最大得分（最大无重复元素和）

变体A | 状态：window_set + cur_sum | 合法：无重复元素

用 set 精确定位重复元素：while 移动 left 直到移除重复的那个，再把新元素纳入。

def maximumUniqueSubarray(nums):
    left, cur_sum, max_sum = 0, nums[0], nums[0]
    window_set = {nums[0]}
    for right in range(1, len(nums)):
        if nums[right] not in window_set:
            cur_sum += nums[right]
            window_set.add(nums[right])
            max_sum = max(max_sum, cur_sum)
        else:
            while nums[left] != nums[right]:   # 收缩到重复元素
                cur_sum -= nums[left]
                window_set.remove(nums[left])
                left += 1
            cur_sum -= nums[left]              # 移除那个重复元素
            window_set.remove(nums[left])
            left += 1
            cur_sum += nums[right]             # 纳入新元素
            window_set.add(nums[right])
    return max_sum

最小覆盖子串

变体B | 状态：need[c] + missing | 合法：missing == 0

need[c] 为负表示多余。合法后收缩掉所有多余字符，记录最短，再强制左移破坏合法性。

def minWindow(s, t):
    need = Counter(t)
    missing = len(t)
    left = 0; best = (float('inf'), 0, 0)
    for right, c in enumerate(s):
        if need[c] > 0: missing -= 1
        need[c] -= 1
        if missing == 0:
            while need[s[left]] < 0:      # 收缩多余字符
                need[s[left]] += 1; left += 1
            best = min(best, (right-left+1, left, right+1))
            need[s[left]] += 1; missing += 1; left += 1  # 强制破坏合法
    return s[best[1]:best[2]] if best[0] < float('inf') else ""

def findAnagrams(s, p):
    need, window, res, left = Counter(p), Counter(), [], 0
    for right, c in enumerate(s):
        window[c] += 1
        if right - left + 1 > len(p):     # 超出固定大小，移出左端
            lc = s[left]; window[lc] -= 1
            if not window[lc]: del window[lc]
            left += 1
        if window == need: res.append(left)
    return res

def solveNQueens(n):
    res = []; cols = set(); diag1 = set(); diag2 = set()
    def backtrack(row, path):
        if row == n:
            res.append(["."*c+"Q"+"."*(n-c-1) for c in path]); return
        for col in range(n):
            if col in cols or (row-col) in diag1 or (row+col) in diag2: continue
            cols.add(col); diag1.add(row-col); diag2.add(row+col)
            path.append(col)
            backtrack(row + 1, path)
            path.pop()
            cols.remove(col); diag1.remove(row-col); diag2.remove(row+col)
    backtrack(0, []); return res

    def check(a, b):
        if not a and not b: return True
        if not a or not b:  return False

        right = dfs(node.right)
        ans = max(ans, left + right)
        return max(left, right) + 1
    dfs(root); return ans

def isBalanced(root):
    def dfs(node):
        if not node: return 0
        left  = dfs(node.left)
        right = dfs(node.right)
        if left == -1 or right == -1 or abs(left - right) > 1: return -1
        return max(left, right) + 1
    return dfs(root) != -1

            if node.right: q.append(node.right)

    return res

    idx_map = {v: i for i, v in enumerate(inorder)}
    def helper(pl, pr, il, ir):
        if pl > pr: return None
        root_val  = preorder[pl]
        idx       = idx_map[root_val]

        node      = TreeNode(root_val)

        return node

打家劫舍 II（环形）

def rob2(nums):
    def rob_linear(arr):
        prev2, prev1 = 0, 0
        for x in arr:
            prev2, prev1 = prev1, max(prev1, prev2 + x)
        return prev1
    n = len(nums)
    if n == 1: return nums[0]
    return max(rob_linear(nums[:-1]), rob_linear(nums[1:]))

最长递增子序列（LIS）

patience sort + 二分，O(n log n)。tails[i] = 长度 i+1 的 LIS 的最小末尾元素，对每个数二分找替换位置。

def lengthOfLIS(nums):
    import bisect
    tails = []
    for num in nums:
        pos = bisect.bisect_left(tails, num)
        if pos == len(tails): tails.append(num)
        else: tails[pos] = num
    return len(tails)

def coinChange(coins, amount):
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    for i in range(1, amount + 1):
        for coin in coins:
            if i >= coin and dp[i-coin] != float('inf'):
                dp[i] = min(dp[i], dp[i-coin] + 1)
    return dp[amount] if dp[amount] != float('inf') else -1

def longestCommonSubsequence(text1, text2):
    m, n = len(text1), len(text2)
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        for j in range(1, n+1):
            if text1[i-1] == text2[j-1]: dp[i][j] = dp[i-1][j-1] + 1
            else: dp[i][j] = max(dp[i-1][j], dp[i][j-1])

def minDistance(word1, word2):
    m, n = len(word1), len(word2)
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i in range(m+1): dp[i][0] = i
    for j in range(n+1): dp[0][j] = j
    for i in range(1, m+1):
        for j in range(1, n+1):
            if word1[i-1] == word2[j-1]: dp[i][j] = dp[i-1][j-1]
            else: dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]

def canPartition(nums):
    total = sum(nums)
    if total % 2 != 0: return False
    target = total // 2
    dp = [False] * (target + 1)
    dp[0] = True
    for num in nums:
        for j in range(target, num-1, -1):   # 逆序！
            dp[j] |= dp[j-num]
    return dp[target]

    n = len(s)
    dp = [[False]*n for _ in range(n)]
    count = 0
    for i in range(n-1, -1, -1):
        for j in range(i, n):
            if s[i] == s[j] and (j-i <= 2 or dp[i+1][j-1]):
                dp[i][j] = True; count += 1
    return count
