#!/usr/bin/env python3
"""Offline release checksum verification; executes no model code."""
import hashlib,json
from pathlib import Path
root=Path(__file__).resolve().parents[1];count=0;expected=set()
for line in (root/'SHA256SUMS').read_text().splitlines():
 want,name=line.split('  ',1);f=root/name
 if not f.resolve().is_relative_to(root):raise SystemExit('Invalid checksum path')
 h=hashlib.sha256()
 with f.open('rb') as s:
  for b in iter(lambda:s.read(16*1024**2),b''):h.update(b)
 if h.hexdigest()!=want:raise SystemExit('Checksum mismatch: '+name)
 expected.add(name);count+=1
# HF-generated control/cache files are not model assets and may be present after download.
actual={str(f.relative_to(root)) for f in root.rglob('*') if f.is_file() and f.name!='SHA256SUMS'
        and not any(x in f.relative_to(root).parts for x in {'.git','.cache','__pycache__'})}
extra=actual-expected
if extra:raise SystemExit('Unexpected files (check for stale v2 artifacts): '+str(sorted(extra)))
print(json.dumps({'passed':True,'files_verified':count},indent=2))
